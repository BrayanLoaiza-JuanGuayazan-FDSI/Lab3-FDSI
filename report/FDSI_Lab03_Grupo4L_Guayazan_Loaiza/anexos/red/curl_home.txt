<script>
const ALERTS_API = '/api/alerts';

async function loadAlerts() {
  document.getElementById('status').textContent = 'actualizando...';
  try {
    const res = await fetch(ALERTS_API + '/alerts');
    const data = await res.json();
    const alerts = data.alerts || [];

    const counts = { CRITICAL: 0, HIGH: 0, MEDIUM: 0, LOW: 0 };
    alerts.forEach(a => { if (counts[a.severity] !== undefined) counts[a.severity]++; });
    document.getElementById('cnt-critical').textContent = counts.CRITICAL;
    document.getElementById('cnt-high').textContent = counts.HIGH;
    document.getElementById('cnt-medium').textContent = counts.MEDIUM;
    document.getElementById('cnt-low').textContent = counts.LOW;

    const tbody = document.getElementById('alerts-body');
    tbody.innerHTML = alerts.map(a => `
      <tr>
        <td class="mono">${a.id}</td>
        <td><span class="badge ${a.severity}">${a.severity}</span></td>
        <td>${a.type}</td>
        <td class="mono">${a.host}</td>
        <td class="mono">${a.user}</td>
        <td><span class="badge ${a.status}">${a.status}</span></td>
        <td class="mono">${a.created_at.replace('T',' ').replace('Z','')}</td>
      </tr>`).join('');

    document.getElementById('status').textContent = `${alerts.length} alertas · ${new Date().toLocaleTimeString()}`;
  } catch(e) {
    document.getElementById('status').textContent = 'Error conectando a la API';
    document.getElementById('alerts-body').innerHTML =
      '<tr><td colspan="7" style="text-align:center;color:#f85149;padding:24px">No se pudo conectar a ' + ALERTS_API + '/alerts</td></tr>';
  }
}

loadAlerts();
setInterval(loadAlerts, 30000);
</script>
